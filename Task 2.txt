STOCK_PRICES = {
    "AAPL": 180,
    "TSLA": 250,
    "MSFT": 420,
    "GOOGL": 175,
    "AMZN": 190
}

def track_portfolio():
    portfolio = {}
    total_investment = 0

    print("=== STOCK PORTFOLIO TRACKER ===")
    print("Available stocks:", ", ".join(STOCK_PRICES))
    print("Enter 'done' when you finish.\n")

    while True:
        stock = input("Enter stock symbol: ").upper().strip()

        if stock == "DONE":
            break

        if stock not in STOCK_PRICES:
            print("Stock not found in the predefined price list.\n")
            continue

        try:
            quantity = int(input(f"Enter quantity of {stock}: "))
            if quantity <= 0:
                print("Quantity must be greater than 0.\n")
                continue
        except ValueError:
            print("Please enter a valid whole number.\n")
            continue

        value = STOCK_PRICES[stock] * quantity
        portfolio[stock] = portfolio.get(stock, 0) + quantity
        total_investment += value

        print(f"{stock}: {quantity} shares = ${value:,.2f}\n")

    print("\n=== PORTFOLIO SUMMARY ===")
    if not portfolio:
        print("No stocks were added.")
        return

    for stock, quantity in portfolio.items():
        value = STOCK_PRICES[stock] * quantity
        print(f"{stock}: {quantity} shares × ${STOCK_PRICES[stock]} = ${value:,.2f}")

    print(f"Total Investment: ${total_investment:,.2f}")

    save = input("Save the result to portfolio.txt? (yes/no): ").lower().strip()
    if save == "yes":
        with open("portfolio.txt", "w") as file:
            file.write("STOCK PORTFOLIO SUMMARY\n")
            file.write("========================\n")
            for stock, quantity in portfolio.items():
                value = STOCK_PRICES[stock] * quantity
                file.write(
                    f"{stock}: {quantity} shares × ${STOCK_PRICES[stock]} "
                    f"= ${value:,.2f}\n"
                )
            file.write(f"Total Investment: ${total_investment:,.2f}\n")
        print("Saved successfully as portfolio.txt")

if __name__ == "__main__":
    track_portfolio()
